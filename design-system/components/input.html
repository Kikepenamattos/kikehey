Hide Section Functionality

Funcionalidad para habilitar/deshabilitar elementos mediante el botón "Hide section". El botón alterna el estado del input y cambia el icono entre eye.svg y eye_hidden.svg.

Ejemplo Funcional

Input name
Hide section
Código de implementación: 
<script>
  // Funcionalidad para Hide Section
  function initHideSection() {
    const hideButtons = document.querySelectorAll('.dsu-input-label__hide');
    
    hideButtons.forEach((hideButton) => {
      hideButton.addEventListener('click', function() {
        // Buscar el grupo de input padre
        const inputGroup = this.closest('.dsu-input-group');
        if (!inputGroup) return;
        
        // Buscar el input dentro del grupo
        const input = inputGroup.querySelector('.dsu-input');
        if (!input) return;
        
        // Buscar el icono para cambiarlo
        const hideIcon = this.querySelector('.dsu-input-label__hide-icon img');
        
        // Toggle del estado disabled
        const isDisabled = input.disabled;
        
        if (isDisabled) {
          // Habilitar el input
          input.disabled = false;
          input.removeAttribute('disabled');
          
          // Cambiar icono de eye_hidden a eye
          if (hideIcon) {
            hideIcon.src = '../../images/eye.svg';
          }
          
          // Remover clase de grupo disabled
          inputGroup.classList.remove('dsu-input-group--hidden');
        } else {
          // Deshabilitar el input
          input.disabled = true;
          input.setAttribute('disabled', 'disabled');
          
          // Cambiar icono de eye a eye_hidden
          if (hideIcon) {
            hideIcon.src = '../../images/eye_hidden.svg';
          }
          
          // Agregar clase al grupo para estilos adicionales
          inputGroup.classList.add('dsu-input-group--hidden');
        }
      });
    });
  }
  
  // Inicializar al cargar
  document.addEventListener('DOMContentLoaded', initHideSection);
</script>

Cómo adaptar para otros elementos

Ejemplo genérico para cualquier elemento: 
// Función genérica para toggle de elementos
function initToggleSection(toggleSelector, targetSelector, iconElement) {
  const toggleButtons = document.querySelectorAll(toggleSelector);
  
  toggleButtons.forEach((toggleButton) => {
    toggleButton.addEventListener('click', function() {
      // Buscar el elemento objetivo
      const targetElement = this.closest('.parent-container')
        .querySelector(targetSelector);
      
      if (!targetElement) return;
      
      // Buscar el icono si se proporciona
      const icon = iconElement 
        ? this.querySelector(iconElement) 
        : null;
      
      // Toggle del estado disabled
      const isDisabled = targetElement.disabled || 
                         targetElement.classList.contains('disabled');
      
      if (isDisabled) {
        // Habilitar el elemento
        targetElement.disabled = false;
        targetElement.removeAttribute('disabled');
        targetElement.classList.remove('disabled');
        
        // Cambiar icono si existe
        if (icon) {
          icon.src = '../../images/eye.svg';
        }
      } else {
        // Deshabilitar el elemento
        targetElement.disabled = true;
        targetElement.setAttribute('disabled', 'disabled');
        targetElement.classList.add('disabled');
        
        // Cambiar icono si existe
        if (icon) {
          icon.src = '../../images/eye_hidden.svg';
        }
      }
    });
  });
}
