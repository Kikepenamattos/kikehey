Dynamic Chips Creation

Ejemplo funcional para crear chips dinámicamente desde un input. Al escribir y presionar Enter, se crea un nuevo chip. Cada chip puede ser eliminado haciendo clic en el botón de cerrar.

Ejemplo interactivo: Escribe y presiona Enter
Código JavaScript completo: 
// Función para crear un chip
function createChip(text) {
    const chip = document.createElement('div');
    chip.className = 'project-chip';
    chip.setAttribute('data-project', text.trim());
    
    const chipText = document.createElement('span');
    chipText.className = 'project-chip__text';
    chipText.textContent = text.trim();
    
    const closeBtn = document.createElement('button');
    closeBtn.className = 'project-chip__close';
    closeBtn.setAttribute('aria-label', 'Remove project');
    closeBtn.innerHTML = '<svg class="project-chip__close-icon" viewBox="0 0 12 12" fill="none" xmlns="http://www.w3.org/2000/svg"><path d="M9 3L3 9M3 3L9 9" stroke="currentColor" stroke-width="1.5" stroke-linecap="round" stroke-linejoin="round"/></svg>';
    
    closeBtn.addEventListener('click', function() {
        chip.remove();
    });
    
    chip.appendChild(chipText);
    chip.appendChild(closeBtn);
    
    return chip;
}

// Input para crear chips
const input = document.getElementById('projectsInput');
const container = document.getElementById('projectsChipsContainer');

input.addEventListener('keypress', function(e) {
    if (e.key === 'Enter') {
        e.preventDefault();
        const value = this.value.trim();
        
        if (value) {
            // Verificar si el chip ya existe
            const existingChips = Array.from(container.querySelectorAll('.project-chip'));
            const exists = existingChips.some(chip => 
                chip.querySelector('.project-chip__text').textContent.toLowerCase() === value.toLowerCase()
            );
            
            if (!exists) {
                const chip = createChip(value);
                container.appendChild(chip);
                this.value = ''; // Limpiar input
            } else {
                alert('Este proyecto ya está agregado');
            }
        }
    }
});
